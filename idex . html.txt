function showPage(id,button){
 document.querySelectorAll(".page").forEach(p=>p.classList.remove("active"));
 document.getElementById(id).classList.add("active");
 document.querySelectorAll(".nav button").forEach(b=>b.classList.remove("active"));
 if(button)button.classList.add("active");
}

function renderCharacters(){
 const grid=document.getElementById("characterGrid");
 grid.innerHTML="";
 characters.forEach(c=>{
  const div=document.createElement("div");
  div.className="character"+(c[0]===selectedCharacter?" selected":"");
  div.innerHTML=`<span class="emoji">${c[1]}</span><b>${c[0]}</b>`;
  div.onclick=()=>selectCharacter(c[0]);
  grid.appendChild(div);
 });
}

function selectCharacter(name){
 selectedCharacter=name;
 const c=characters.find(x=>x[0]===name);
 document.getElementById("sideCharacter").textContent=name+" "+c[1];
 document.getElementById("chatCharacter").textContent=name+" ♡";
 document.documentElement.style.setProperty("--accent",c[2]);
 document.documentElement.style.setProperty("--accent2",c[3]);
 document.getElementById("accentPicker").value=c[2];
 document.getElementById("secondaryPicker").value=c[3];
 localStorage.setItem("yoruCharacter",name);
 renderCharacters();
}

function randomCharacter(){
 const c=characters[Math.floor(Math.random()*characters.length)];
 selectCharacter(c[0]);
}

function applyTheme(){
 const a=document.getElementById("accentPicker").value;
 const s=document.getElementById("secondaryPicker").value;
 const b=document.getElementById("backgroundPicker").value;
 document.documentElement.style.setProperty("--accent",a);
 document.documentElement.style.setProperty("--accent2",s);
 document.documentElement.style.setProperty("--bg",b);
 document.body.style.background=b;
 localStorage.setItem("yoruTheme",JSON.stringify({a,s,b}));
}

function resetTheme(){
 localStorage.removeItem("yoruTheme");
 document.documentElement.style.setProperty("--accent","#d99a64");
 document.documentElement.style.setProperty("--accent2","#e9a7a0");
 document.body.style.background="";
 document.getElementById("accentPicker").value="#d99a64";
 document.getElementById("secondaryPicker").value="#e9a7a0";
 document.getElementById("backgroundPicker").value="#100b0d";
}

function sendMessage(){
 const input=document.getElementById("chatInput");
 const value=input.value.trim();
 if(!value)return;
 const messages=JSON.parse(localStorage.getItem("yoruChat")||"[]");
 messages.push({text:value,time:new Date().toLocaleTimeString()});
 localStorage.setItem("yoruChat",JSON.stringify(messages));
 input.value="";
 renderChat();
}

function renderChat(){
 const box=document.getElementById("chatMessages");
 const messages=JSON.parse(localStorage.getItem("yoruChat")||"[]");
 box.innerHTML='<div class="message">Welcome to our little room ♡<small>System</small></div>';
 messages.forEach(m=>{
  const div=document.createElement("div");
  div.className="message mine";
  const content=document.createTextNode(m.text);
  const time=document.createElement("small");
  time.textContent=m.time;
  div.append(content,time);
  box.appendChild(div);
 });
 box.scrollTop=box.scrollHeight;
}

function saveLetter(){
 const title=document.getElementById("letterTitle").value.trim();
 const body=document.getElementById("letterText").value.trim();
 if(!title||!body)return alert("Write your letter first ♡");
 const list=JSON.parse(localStorage.getItem("yoruLetters")||"[]");
 list.unshift({title,body,date:new Date().toLocaleDateString()});
 localStorage.setItem("yoruLetters",JSON.stringify(list));
 document.getElementById("letterTitle").value="";
 document.getElementById("letterText").value="";
 renderLetters();
}

function renderLetters(){
 const list=JSON.parse(localStorage.getItem("yoruLetters")||"[]");
 document.getElementById("letterList").innerHTML=list.map(l=>
 `<div class="panel"><h3>${escapeHTML(l.title)}</h3><p>${escapeHTML(l.body).replace(/\n/g,"<br>")}</p><small>${l.date}</small></div>`
 ).join("");
}

function saveMemory(){
 const file=document.getElementById("memoryFile").files[0];
 const caption=document.getElementById("memoryCaption").value.trim();
 if(!file)return alert("Choose a picture first.");
 const reader=new FileReader();
 reader.onload=()=>{
  const list=JSON.parse(localStorage.getItem("yoruMemories")||"[]");
  list.unshift({image:reader.result,caption});
  try{
   localStorage.setItem("yoruMemories",JSON.stringify(list));
   renderMemories();
  }catch(e){alert("Storage is full. Try a smaller image.");}
 };
 reader.readAsDataURL(file);
}

function renderMemories(){

 document.getElementById("memoryList").innerHTML=list.map(m=>

 ).join("");
}

function saveTimeline(){

 list.unshift({date,story});
 localStorage.setItem("yoruTimeline",JSON.stringify(list));
 document.getElementById("timelineDate").value="";
 document.getElementById("timelineText").value="";
 renderTimeline();
}

function renderTimeline(){

function escapeHTML(value){
 return String(value).replace(/[&<>"']/g,c=>({
 "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
 }[c]));
}

document.getElementById("characterUpload").addEventListener("change",e=>{
 const file=e.target.files[0];
 if(!file)return;
 const reader=new FileReader();
 reader.onload=()=>{
  customCharacterImage=reader.result;
  const preview=document.getElementById("characterPreview");
  preview.src=customCharacterImage;
  preview.hidden=false;
  localStorage.setItem("yoruCustomCharacter",customCharacterImage);
 };
 reader.readAsDataURL(file);
});
